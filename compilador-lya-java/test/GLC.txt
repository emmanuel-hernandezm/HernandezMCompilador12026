G = (Vn, Vt, S, P)



Vn(No terminales):

<Programa>
<Bloque>
<Seccion_Const>
<Lista_Const>
<Lista_Const_Resto>
<Seccion_Var>
<Lista_Var>
<Lista_Var_Resto>
<Seccion_Proced>
<Lista_Proced>
<Lista_Proced_Resto>
<Decl_Proced>
<Proposición>
<Lista_Proposiciones>
<Lista_Proposiciones_Resto>
<Expresión>
<Expresión_Resto>
<Término>
<Término_Resto>
<Factor>
<Condición>
<Operador_Relacional>
<Argumento_Write>
<Direccion_For>




Vt(Terminales):

id, num, const, var, proced, begin, end, write, read, call, if, then, while, do,
for, to, down, =, ==, !=, <, >, <=, >=, +, -, *, /, (, ), ., ",", ;




S(Simbolo Inicial): Programa




P(Producciones):

<Programa> -> <Bloque> "."

<Bloque> -> <Seccion_Const> <Seccion_Var> <Seccion_Proced> <Proposición>

<Seccion_Const> -> "const" <Lista_Const> ";"
<Seccion_Const> -> ε

<Lista_Const> -> id "=" num <Lista_Const_Resto>
<Lista_Const_Resto> -> "," id "=" num <Lista_Const_Resto>
<Lista_Const_Resto> -> ε

<Seccion_Var> -> "var" <Lista_Var> ";"
<Seccion_Var> -> ε

<Lista_Var> -> id <Lista_Var_Resto>

<Lista_Var_Resto> -> "," id <Lista_Var_Resto>
<Lista_Var_Resto> -> ε

<Seccion_Proced> -> "proced" id ";" <Bloque> ";" <Seccion_Proced>
<Seccion_Proced> -> ε

<Proposición> -> "begin" <Lista_Proposiciones> "end"
<Proposición> -> id "=" <Expresión>
<Proposición> -> "write" <Argumento_Write>
<Proposición> -> "read" id
<Proposición> -> "call" id
<Proposición> -> "if" <Condición> "then" <Proposición>
<Proposición> -> "while" <Condición> "do" <Proposición>
<Proposición> -> "for" id "=" <Expresión> <Direccion_For> <Expresión> "do" <Proposición>

<Lista_Proposiciones> -> <Proposición> <Lista_Proposiciones_Resto>

<Lista_Proposiciones_Resto> -> ";" <Proposición> <Lista_Proposiciones_Resto>
<Lista_Proposiciones_Resto> -> ε

<Expresión> -> <Término> <Expresión_Resto>

<Expresión_Resto> -> "+" <Término> <Expresión_Resto>
<Expresión_Resto> -> "-" <Término> <Expresión_Resto>
<Expresión_Resto> -> ε

<Término> -> <Factor> <Término_Resto>

<Término_Resto> -> "*" <Factor> <Término_Resto>
<Término_Resto> -> "/" <Factor> <Término_Resto>
<Término_Resto> -> ε

<Factor> -> "(" <Expresión> ")"
<Factor> -> id
<Factor> -> num

<Condición> -> <Expresión> <Operador_Relacional> <Expresión>

<Operador_Relacional> -> "=="
<Operador_Relacional> -> "!="
<Operador_Relacional> -> "<"
<Operador_Relacional> -> ">"
<Operador_Relacional> -> "<="
<Operador_Relacional> -> ">="

<Argumento_Write> -> id
<Argumento_Write> -> num

<Direccion_For> -> "to"
<Direccion_For> -> "down"
